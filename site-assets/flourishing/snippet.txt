  function dotOf(d) { return typeof d.icon === 'number' ? ICONS[d.icon] : esc(d.n); }
  function kindOf(d) { return d.kind === 'retreat' ? 'retreat' : d.kind === 'tbd' ? 'tbd' : ''; }

    var btns = map.querySelectorAll('.stop'), card = sec.querySelector('.map-card'), list = sec.querySelector('.trail');
    function pick(i) {
      var d = stops[i]; if (!d || !card) return;
      for (var j = 0; j < btns.length; j++) btns[j].setAttribute('aria-pressed', j === i ? 'true' : 'false');
      var k = kindOf(d);
      card.className = 'map-card' + (k ? ' ' + k : '');
      var rows = d.kind === 'tbd'
        ? '<dl><dt>Status</dt><dd>This module is in the workbook draft. Its title and summary go here once it\u2019s final.</dd></dl>'

    }
    Array.prototype.forEach.call(btns, function (b, i) {

      b.addEventListener('focus', function () { pick(i); });
      b.addEventListener('click', function () { pick(i); });
    });
    pick(0);
